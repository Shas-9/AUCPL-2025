#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n, m;
    cin >> n >> m;

    if (n % 2) {
        cout << "impossible\n";
        return 0;
    }

    vector<vector<int>> g(n + 1);
    for (int i = 0; i < m; i++) {
        int a, b;
        cin >> a >> b;
        g[a].push_back(b);
        g[b].push_back(a);
    }

    for (int i = 1; i <= n; i++)
        sort(g[i].begin(), g[i].end());

    set<int> unused;
    for (int i = 1; i <= n; i++) unused.insert(i);

    vector<int> color(n + 1, -1);
    vector<pair<int,int>> comps;
    vector<vector<int>> nodes;

    queue<int> q;

    while (!unused.empty()) {
        int start = *unused.begin();
        unused.erase(start);

        q.push(start);
        color[start] = 0;

        int c0 = 1, c1 = 0;
        vector<int> comp = {start};

        while (!q.empty()) {
            int u = q.front(); q.pop();

            vector<int> to_visit;

            for (auto it = unused.begin(); it != unused.end();) {
                int v = *it;

                if (!binary_search(g[u].begin(), g[u].end(), v)) {
                    to_visit.push_back(v);
                    it = unused.erase(it);
                } else {
                    ++it;
                }
            }

            for (int v : to_visit) {
                color[v] = color[u] ^ 1;
                if (color[v]) c1++;
                else c0++;

                q.push(v);
                comp.push_back(v);
            }
        }

        comps.push_back({c0,c1});
        nodes.push_back(comp);
    }

    int k = comps.size();
    int target = n / 2;

    vector<int> dp(target + 1, -1);
    dp[0] = -2;

    for (int i = 0; i < k; i++) {
        int a = comps[i].first;
        int b = comps[i].second;

        vector<int> ndp = dp;

        for (int j = 0; j <= target; j++) {
            if (dp[j] != -1) {
                if (j + a <= target) ndp[j + a] = i;
                if (j + b <= target) ndp[j + b] = i;
            }
        }

        dp.swap(ndp);
    }

    if (dp[target] == -1) {
        cout << "impossible\n";
        return 0;
    }

    vector<int> choose(k);
    int cur = target;

    for (int i = k - 1; i >= 0; i--) {
        int a = comps[i].first;
        int b = comps[i].second;

        if (cur - a >= 0 && dp[cur - a] != -1) {
            choose[i] = 0;
            cur -= a;
        } else {
            choose[i] = 1;
            cur -= b;
        }
    }

    vector<char> ans(n + 1);

    for (int i = 0; i < k; i++) {
        for (int v : nodes[i]) {
            int c = color[v];
            if (choose[i]) c ^= 1;
            ans[v] = (c ? 'b' : 'r');
        }
    }

    for (int i = 1; i <= n; i++)
        cout << ans[i] << "\n";
}
