#include <bits/stdc++.h>
using namespace std;

using pii = pair<int,int>;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n, m;
    cin >> n >> m;

    vector<string> grid(n);
    for (int i = 0; i < n; i++) cin >> grid[i];

    vector<vector<int>> sand(n, vector<int>(m, INT_MAX));
    queue<pii> q;

    pii start, exitPos;

    // Initialize
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            if (grid[i][j] == '*') {
                sand[i][j] = 0;
                q.push({i, j});
            }
            if (grid[i][j] == 'M') start = {i, j};
            if (grid[i][j] == 'E') exitPos = {i, j};
        }
    }

    int dr[4] = {1, -1, 0, 0};
    int dc[4] = {0, 0, 1, -1};

    
    while (!q.empty()) {
        auto [r, c] = q.front(); q.pop();

        for (int d = 0; d < 4; d++) {
            int nr = r + dr[d];
            int nc = c + dc[d];

            if (nr < 0 || nr >= n || nc < 0 || nc >= m) continue;
            if (grid[nr][nc] == '#') continue;

            if (sand[nr][nc] > sand[r][c] + 1) {
                sand[nr][nc] = sand[r][c] + 1;
                q.push({nr, nc});
            }
        }
    }

    
    vector<vector<int>> dist(n, vector<int>(m, -1));
    queue<pii> mq;

    if (sand[start.first][start.second] == 0) {
        cout << -1 << "\n";
        return 0;
    }

    dist[start.first][start.second] = 0;
    mq.push(start);

    while (!mq.empty()) {
        auto [r, c] = mq.front(); mq.pop();
        int t = dist[r][c];

        // Reached exit
        if (make_pair(r, c) == exitPos) {
            cout << t << "\n";
            return 0;
        }

        for (int d = 0; d < 4; d++) {
            int nr = r + dr[d];
            int nc = c + dc[d];
            int nt = t + 1;

            if (nr < 0 || nr >= n || nc < 0 || nc >= m) continue;
            if (grid[nr][nc] == '#') continue;
            if (dist[nr][nc] != -1) continue;

            
            if (sand[nr][nc] <= nt) continue;

            dist[nr][nc] = nt;
            mq.push({nr, nc});
        }
    }

    cout << -1 << "\n";
    return 0;
}
