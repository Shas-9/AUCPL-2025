#include <bits/stdc++.h>
using namespace std;

// KMP: find all occurrences of pattern in text, return start indices
vector<int> kmpSearch(const string& text, const string& pattern) {
    int n = text.size(), m = pattern.size();
    vector<int> occ;
    if (m > n) return occ;

    // Build failure function
    vector<int> fail(m, 0);
    for (int i = 1; i < m; i++) {
        int j = fail[i - 1];
        while (j > 0 && pattern[i] != pattern[j]) j = fail[j - 1];
        if (pattern[i] == pattern[j]) j++;
        fail[i] = j;
    }

    // Search
    int j = 0;
    for (int i = 0; i < n; i++) {
        while (j > 0 && text[i] != pattern[j]) j = fail[j - 1];
        if (text[i] == pattern[j]) j++;
        if (j == m) {
            occ.push_back(i - m + 1);
            j = fail[j - 1];
        }
    }
    return occ;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n, m, k;
    cin >> n >> m >> k;

    string files, name;
    cin >> files >> name;

    vector<int> pos = kmpSearch(files, name);
    int total = (int)pos.size();

    if (total < k) {
        cout << "impossible\n";
        return 0;
    }

    // Sliding window of k consecutive occurrences
    // Minimum redaction length for window [i, i+k-1]:
    //   max(1, pos[i+k-1] - pos[i] - m + 2)
    int ans = INT_MAX;
    for (int i = 0; i + k - 1 < total; i++) {
        int len = pos[i + k - 1] - pos[i] - m + 2;
        ans = min(ans, max(1, len));
    }

    cout << ans << "\n";
    return 0;
}
