#include <bits/stdc++.h>
using namespace std;

int main(){
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    
    int n, m;
    cin >> n >> m;
    
    map<string,int> taskIdx;
    for(int i = 0; i < n; i++){
        string s; cin >> s;
        taskIdx[s] = i;
    }
    
    vector<int> emp(m);
    for(int i = 0; i < m; i++){
        int k; cin >> k;
        int mask = 0;
        for(int j = 0; j < k; j++){
            string s; cin >> s;
            mask |= (1 << taskIdx[s]);
        }
        emp[i] = mask;
    }
    
    int full = (1 << n) - 1;
    vector<int> dp(1 << n, INT_MAX);
    dp[0] = 0;
    
    for(int mask = 0; mask <= full; mask++){
        if(dp[mask] == INT_MAX) continue;
        for(int i = 0; i < m; i++){
            int nmask = mask | emp[i];
            if(dp[nmask] > dp[mask] + 1)
                dp[nmask] = dp[mask] + 1;
        }
    }
    
    cout << dp[full] << "\n";
    return 0;
}
