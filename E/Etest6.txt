#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n;
    cin >> n;

    vector<int> v(n);
    int p = 0;
    int ops = 0;

    while (ops < 125) {
        cin >> v[0];
        for (int i = 1; i < n; i++) cin >> v[i];
        cin >> p;

        if (p >= 70) break;

        int l = -1;

        for (int i = 0; i < n; i++) {
            if (v[i] == 0) {
                l = i;
                break;
            }
        }

        if (l == -1) break;

        int r = l;
        while (r + 1 < n && v[r + 1] == 0) r++;

        cout << l + 1 << " " << r + 1 << endl;
        cout.flush();

        ops++;
    }

    return 0;
}
