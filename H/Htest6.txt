#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    string s1, s2, s3;
    cin >> s1 >> s2 >> s3;

    int i1 = 0, i2 = 0, i3 = 0;
    string res;

    while(i1 < s1.size() || i2 < s2.size() || i3 < s3.size()) {
        set<char> possible = {'1','2','3'};

        if(i1 < s1.size()) possible.erase(s1[i1]);
        if(i2 < s2.size()) possible.erase(s2[i2]);
        if(i3 < s3.size()) possible.erase(s3[i3]);

        // Exactly one winner remains
        char winner = *possible.begin();
        res.push_back(winner);

        // Advance all pointers where the winner is next
        if(i1 < s1.size() && s1[i1] == winner) i1++;
        if(i2 < s2.size() && s2[i2] == winner) i2++;
        if(i3 < s3.size() && s3[i3] == winner) i3++;
    }

    cout << res << "\n";
}
