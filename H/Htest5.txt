#include <bits/stdc++.h>
using namespace std;

int main() {
    string s1, s2, s3;
    cin >> s1 >> s2 >> s3;

    int p1 = 0, p2 = 0, p3 = 0;
    string res;

    int n = s1.size() + 1; // total games = length of any list + 1
    for (int i = 0; i < n; i++) {
        char next;
        if (p1 < s1.size() && (p2 >= s2.size() || s1[p1] != '1') && (p3 >= s3.size() || s1[p1] != '1')) {
            next = '1';
        } else if (p2 < s2.size() && (p1 >= s1.size() || s2[p2] != '2') && (p3 >= s3.size() || s2[p2] != '2')) {
            next = '2';
        } else {
            next = '3';
        }
        res += next;
        if (p1 < s1.size() && s1[p1] == next) p1++;
        if (p2 < s2.size() && s2[p2] == next) p2++;
        if (p3 < s3.size() && s3[p3] == next) p3++;
    }

    cout << res << "\n";
    return 0;
}
