#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

int main(){
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    
    int n, m;
    cin >> n >> m;
    
    vector<string> grid(n);
    for(int i = 0; i < n; i++) cin >> grid[i];
    
    // dirt[j] = number of D's in column j
    vector<ll> dirt(m, 0);
    for(int i = 0; i < n; i++)
        for(int j = 0; j < m; j++)
            if(grid[i][j] == 'D') dirt[j]++;
    
    ll total_dirt = 0;
    for(int j = 0; j < m; j++) total_dirt += dirt[j];
    
    // Prefix sum: P[j] = sum of dirt[0..j-1]
    vector<ll> P(m+1, 0);
    for(int j = 0; j < m; j++) P[j+1] = P[j] + dirt[j];
    
    // cost(l,r) = (r-l+1)*n - 2*(P[r+1]-P[l]) + total_dirt
    // For fixed r, minimize over l in [0,r]: (-n*l + 2*P[l])
    // Then cost = (n*r - 2*P[r+1] + n + total_dirt) + min_val
    
    ll ans = LLONG_MAX;
    ll min_val = 0; // at l=0: -n*0 + 2*P[0] = 0
    
    for(int r = 0; r < m; r++){
        // Update min_val to include l = r (so l can equal r)
        ll lval = -(ll)n * r + 2 * P[r];
        min_val = min(min_val, lval);
        
        ll cost = (ll)n * r - 2 * P[r+1] + n + total_dirt + min_val;
        ans = min(ans, cost);
    }
    
    cout << ans * 1000 << "\n";
    return 0;
}
