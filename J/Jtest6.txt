#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n;
    cin >> n;

    string prev = "";
    vector<string> ans(n);

    for (int i = 0; i < n; i++) {
        int p;
        cin >> p;

        vector<string> parts(p);
        for (auto &s : parts) cin >> s;

        sort(parts.begin(), parts.end());

        auto it = lower_bound(parts.begin(), parts.end(), prev);

        if (it == parts.end()) {
            cout << "impossible\n";
            return 0;
        }

        ans[i] = *it;
        prev = *it;
    }

    for (auto &s : ans)
        cout << s << "\n";
}
