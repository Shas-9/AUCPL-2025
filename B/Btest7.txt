#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    long long d;
    long long a, o;
    long long da, do_;

    cin >> d;
    cin >> a >> o;
    cin >> da >> do_;

    long long A = a - d * da;
    long long O = o - d * do_;

    long double result = 100.0L * (long double)A / (A + O);

    cout << fixed << setprecision(15) << result << "\n";

    return 0;
}
