#include <bits/stdc++.h>
using namespace std;

int parent[1000005];
long long sz[1000005];

int find(int x) {
    if (parent[x] != x) parent[x] = find(parent[x]);
    return parent[x];
}

void unite(int a, int b) {
    a = find(a); b = find(b);
    if (a == b) return;
    if (sz[a] < sz[b]) swap(a, b);
    parent[b] = a;
    sz[a] += sz[b];
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    
    int R, C, Q;
    cin >> R >> C >> Q;
    
    for (int i = 0; i < R * C; i++) {
        parent[i] = i;
        sz[i] = 100;
    }
    
    auto idx = [&](int r, int c) { return (r-1)*C + (c-1); };
    
    while (Q--) {
        int r1, c1, r2, c2;
        cin >> r1 >> c1 >> r2 >> c2;
        unite(idx(r1,c1), idx(r2,c2));
        cout << sz[find(idx(r1,c1))] << "\n";
    }
}
